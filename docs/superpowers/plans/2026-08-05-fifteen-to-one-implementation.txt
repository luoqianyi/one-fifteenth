# 十五分之一实施计划

> **执行代理必读：** 必须使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans，逐任务执行本计划。使用 `- [ ]` 复选框跟踪步骤。

**目标：** 构建一个可直接部署到 GitHub Pages 的纯前端输入输出学习网站，支持多领域词库、AI 关键词建议、间隔重复抽卡、双阶段计时、本地录音、学习档案和 JSON 备份。

**架构：** 应用采用原生 ES Modules 的单页架构。轻量偏好和活动计时器保存在 localStorage，领域、类目、关键词、训练记录和录音保存在 IndexedDB；纯函数核心用 Node 内置测试，浏览器集成流程用 Playwright 验证。

**技术栈：** HTML5、CSS3、原生 JavaScript、IndexedDB、MediaRecorder、Web Notifications、OpenAI 兼容 Chat Completions API、Node test runner、Playwright。

`scripts/storage.js` 对外接口固定为：

```js
export const DB_NAME = 'fifteen-to-one';
export const DB_VERSION = 1;
export const STORES = ['domains', 'categories', 'keywords', 'sessions', 'recordings'];

export function requestToPromise(request) {
  return new Promise((resolve, reject) => {
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
}

function addIndex(store, name) {
  if (!store.indexNames.contains(name)) store.createIndex(name, name, { unique: false });
}

export function openDatabase() {
  return new Promise((resolve, reject) => {
    const request = indexedDB.open(DB_NAME, DB_VERSION);
    request.onupgradeneeded = () => {
      const db = request.result;
      const transaction = request.transaction;
      const getStore = name => db.objectStoreNames.contains(name)
        ? transaction.objectStore(name)
        : db.createObjectStore(name, { keyPath: 'id' });
      getStore('domains');
      const categories = getStore('categories');
      addIndex(categories, 'domainId');
      const keywords = getStore('keywords');
      ['domainId', 'categoryId', 'lastStudiedAt'].forEach(name => addIndex(keywords, name));
      const sessions = getStore('sessions');
      ['keywordId', 'domainId', 'completedAt'].forEach(name => addIndex(sessions, name));
      const recordings = getStore('recordings');
      addIndex(recordings, 'sessionId');
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
    request.onblocked = () => reject(new Error('数据库升级被其他页面阻止'));
  });
}

export async function withStore(storeNames, mode, operation) {
  const db = await openDatabase();
  const names = Array.isArray(storeNames) ? storeNames : [storeNames];
  return new Promise((resolve, reject) => {
    const transaction = db.transaction(names, mode);
    const stores = Object.fromEntries(names.map(name => [name, transaction.objectStore(name)]));
    let result;
    try {
      result = operation(names.length === 1 ? stores[names[0]] : stores, transaction);
    } catch (error) {
      transaction.abort();
      reject(error);
      return;
    }
    transaction.oncomplete = async () => {
      try { resolve(await result); } catch (error) { reject(error); } finally { db.close(); }
    };
    transaction.onerror = () => { db.close(); reject(transaction.error); };
    transaction.onabort = () => { db.close(); reject(transaction.error ?? new Error('数据库事务已取消')); };
  });
}
```

对象仓储使用 `keyPath: 'id'`。索引：categories 建 `domainId`；keywords 建 `domainId`、`categoryId`、`lastStudiedAt`；sessions 建 `keywordId`、`domainId`、`completedAt`；recordings 建 `sessionId`。

`scripts/services/preferences.js`：

```js
const KEY = 'fifteen-to-one:preferences';
const defaults = {
  activeDomainId: 'all', apiBaseUrl: 'https://api.openai.com/v1',
  apiKey: '', apiModel: 'gpt-4.1-mini', soundEnabled: true,
  notificationsEnabled: false
};

export function loadPreferences() {
  try {
    const stored = JSON.parse(localStorage.getItem(KEY) ?? '{}');
    return { ...defaults, ...stored };
  } catch {
    return { ...defaults };
  }
}

export function savePreferences(patch) {
  const next = { ...loadPreferences(), ...patch };
  localStorage.setItem(KEY, JSON.stringify(next));
  return next;
}

export function exportSafePreferences() {
  const { apiKey, ...safe } = loadPreferences();
  return safe;
}
```

```js
const LEVELS = new Set(['beginner', 'advanced', 'comprehensive']);

export function normalizeName(value) {
  return String(value).trim().replace(/\s+/g, ' ').toLocaleLowerCase('zh-CN');
}

export function parseSuggestions(content) {
  const clean = String(content).replace(/^```(?:json)?\s*/i, '').replace(/\s*```$/i, '');
  const start = clean.indexOf('{');
  const end = clean.lastIndexOf('}');
  if (start < 0 || end <= start) throw new Error('AI 未返回可解析的 JSON');
  const parsed = JSON.parse(clean.slice(start, end + 1));
  const suggestions = Array.isArray(parsed.suggestions) ? parsed.suggestions : [];
  const valid = suggestions.flatMap(item => {
    const name = typeof item?.name === 'string' ? item.name.trim().slice(0, 80) : '';
    const summary = typeof item?.summary === 'string' ? item.summary.trim().slice(0, 240) : '';
    return name && LEVELS.has(item?.level) ? [{ name, level: item.level, summary }] : [];
  });
  if (!valid.length) throw new Error('没有有效的关键词');
  return valid;
}

export function removeDuplicateSuggestions(suggestions, existingNames) {
  const seen = new Set(existingNames.map(normalizeName));
  return suggestions.filter(item => {
    const key = normalizeName(item.name);
    if (!key || seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

      content: `为学习领域“${domain}”的类目“${category}”生成 ${count} 个值得学习的关键词。输出格式：{"suggestions":[{"name":"关键词","level":"beginner","summary":"不超过80字的简介"}]}`

export async function requestSuggestions(config, input, signal) {
  try {
    const response = await fetch(endpoint(config.apiBaseUrl), {
      method: 'POST',
      signal,
      headers: {
        Authorization: `Bearer ${config.apiKey}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: config.apiModel,
        temperature: 0.4,
        messages: buildSuggestionMessages(input)
      })
    });
    const body = await response.json().catch(() => ({}));
    if (!response.ok) {
      throw new Error(body?.error?.message || `AI 接口返回 HTTP ${response.status}`);
    }
    const content = body?.choices?.[0]?.message?.content;
    if (typeof content !== 'string') throw new Error('AI 响应缺少文本内容');
    return parseSuggestions(content);
  } catch (error) {
    if (error instanceof TypeError) throw new Error('跨域或网络连接失败，请检查接口是否允许浏览器直连');
    throw error;
  }
}
```

```js
export function calculateWeight(keyword, now = Date.now()) {
  if (!keyword.studyCount) return 12;
  const hours = Math.max(0, now - Date.parse(keyword.lastStudiedAt)) / 3_600_000;
  const spacing = Math.min(8, 1 + Math.log2(1 + hours / 24));
  const mastery = Math.max(1, 6 - Number(keyword.mastery || 0));
  const recentDrawPenalty = keyword.lastDrawnAt && now - Date.parse(keyword.lastDrawnAt) < 3_600_000 ? 0.2 : 1;
  const skipPenalty = Math.max(0.6, 1 - Number(keyword.skipCount || 0) * 0.05);
  return Math.max(0.1, spacing * mastery * recentDrawPenalty * skipPenalty);
}

export function chooseWeighted(weightedItems, random = Math.random) {
  if (!weightedItems.length) return null;
  const total = weightedItems.reduce((sum, item) => sum + item.weight, 0);
  let point = random() * total;
  for (const item of weightedItems) {
    point -= item.weight;
    if (point <= 0) return item;
  }
  return weightedItems.at(-1);
}

export function drawKeyword(keywords, random = Math.random, now = Date.now()) {
  return chooseWeighted(keywords.map(keyword => ({
    ...keyword, weight: calculateWeight(keyword, now)
  })), random);
}
```

```js
const MIME_TYPES = ['audio/webm;codecs=opus', 'audio/webm', 'audio/mp4'];

export class RecorderService {
  constructor(mediaDevices = navigator.mediaDevices) {
    this.mediaDevices = mediaDevices;
    this.stream = null;
    this.recorder = null;
    this.chunks = [];
    this.startedAt = null;
  }

  async start() {
    if (!this.mediaDevices?.getUserMedia || typeof MediaRecorder === 'undefined') {
      throw new Error('当前浏览器不支持录音');
    }
    this.stream = await this.mediaDevices.getUserMedia({ audio: true });
    const mimeType = MIME_TYPES.find(type => MediaRecorder.isTypeSupported(type)) || '';
    this.chunks = [];
    this.recorder = new MediaRecorder(this.stream, mimeType ? { mimeType } : undefined);
    this.recorder.ondataavailable = event => {
      if (event.data.size) this.chunks.push(event.data);
    };
    this.startedAt = Date.now();
    this.recorder.start();
  }

  stop() {
    if (!this.recorder || this.recorder.state === 'inactive') {
      return Promise.reject(new Error('当前没有正在进行的录音'));
    }
    return new Promise((resolve, reject) => {
      const recorder = this.recorder;
      recorder.onerror = event => {
        this.releaseTracks();
        reject(event.error || new Error('录音失败'));
      };
      recorder.onstop = () => {
        const mimeType = recorder.mimeType || 'audio/webm';
        const result = {
          blob: new Blob(this.chunks, { type: mimeType }),
          mimeType,
          durationMs: Date.now() - this.startedAt
        };
        this.releaseTracks();
        resolve(result);
      };
      recorder.stop();
    });
  }

  cancel() {
    if (this.recorder?.state !== 'inactive') this.recorder.stop();
    this.releaseTracks();
    this.chunks = [];
  }

  releaseTracks() {
    this.stream?.getTracks().forEach(track => track.stop());
    this.stream = null;
    this.recorder = null;
    this.startedAt = null;
  }

  get isRecording() {
    return this.recorder?.state === 'recording';
  }
}
```

MIME 优先顺序为 `audio/webm;codecs=opus`、`audio/webm`、`audio/mp4`。所有退出路径必须执行 `stream.getTracks().forEach(track => track.stop())`。

```js
export const BACKUP_VERSION = 1;
const COLLECTIONS = ['domains', 'categories', 'keywords', 'sessions', 'recordings'];

export function blobToDataUrl(blob) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(reader.result);
    reader.onerror = () => reject(reader.error);
    reader.readAsDataURL(blob);
  });
}

export function dataUrlToBlob(dataUrl) {
  if (!String(dataUrl).startsWith('data:audio/')) throw new Error('录音数据格式无效');
  const [metadata, encoded] = dataUrl.split(',', 2);
  const mimeType = metadata.slice(5).split(';')[0];
  const bytes = Uint8Array.from(atob(encoded), character => character.charCodeAt(0));
  return new Blob([bytes], { type: mimeType });
}

export function createBackupDocument(data, exportedAt = new Date().toISOString()) {
  const { apiKey, ...preferences } = data.preferences ?? {};
  return {
    schemaVersion: BACKUP_VERSION,
    exportedAt,
    preferences,
    domains: data.domains ?? [],
    categories: data.categories ?? [],
    keywords: data.keywords ?? [],
    sessions: data.sessions ?? [],
    recordings: data.recordings ?? []
  };
}

function requireIds(items, label) {
  if (!Array.isArray(items)) throw new Error(`${label} 必须是数组`);
  if (items.some(item => typeof item?.id !== 'string' || !item.id)) {
    throw new Error(`${label} 包含无效 ID`);
  }
}

export function validateBackup(document) {
  if (!document || document.schemaVersion !== BACKUP_VERSION) {
    throw new Error('不支持的备份版本');
  }
  for (const collection of COLLECTIONS) requireIds(document[collection], collection);
  const domainIds = new Set(document.domains.map(item => item.id));
  const categoryIds = new Set(document.categories.map(item => item.id));
  const sessionIds = new Set(document.sessions.map(item => item.id));
  if (document.categories.some(item => !domainIds.has(item.domainId))) {
    throw new Error('备份包含无效的类目领域关系');
  }
  if (document.keywords.some(item => !domainIds.has(item.domainId) || !categoryIds.has(item.categoryId))) {
    throw new Error('备份包含无效的关键词关系');
  }
  if (document.sessions.some(item => item.mastery < 1 || item.mastery > 5 || !item.keywordSnapshot)) {
    throw new Error('备份包含无效的训练记录');
  }
  if (document.recordings.some(item => !sessionIds.has(item.sessionId) || !String(item.dataUrl).startsWith('data:audio/'))) {
    throw new Error('备份包含无效的录音记录');
  }
  return document;
}

export function estimateBackupBytes(document) {
  return new Blob([JSON.stringify(document)]).size;
}
```

备份顶层严格包含 `schemaVersion`、`exportedAt`、`preferences`、`domains`、`categories`、`keywords`、`sessions`、`recordings`。录音项保存 id、sessionId、mimeType 和 dataUrl。校验所有集合均为数组、ID 为非空字符串、外键能解析、掌握度在 1–5、dataUrl 以 `data:audio/` 开头。任何失败在写数据库前抛出。

导出前读取全部数据、转换录音、显示预计大小，确认后创建下载链接 `十五分之一-YYYY-MM-DD.json`。导入选择文件后先解析和校验，展示数量摘要，再选择“合并”或“覆盖”。

合并时冲突 ID 生成新 UUID，并同步重写 domainId、categoryId、keywordId、sessionId 和 recordingId；覆盖时在同一个跨 store 读写事务中 clear 后 put。API Key 不从导入文件更新。
